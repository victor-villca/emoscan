sequenceDiagram
    %% Título simple (el estilo se aplica externamente si es posible)
    title Captura, detección y selección de rostros

    %% Participantes (Alias simplificados, \n para saltos de línea si el renderizador lo soporta en alias)
    %% Nota: El soporte de HTML/Markdown en alias puede variar. Nombres simples son más seguros.
    participant Usuario
    participant Web as Aplicación Web
    participant Plataforma as Plataforma de videollamadas
    participant API as Backend API
    participant Servicio as Servicio de Detección de Rostros
    participant DB as Base de Datos

    %% Activación para mejor visualización
    activate Usuario

    %% Flujo
    Usuario->>Web: Clic en "Capturar pantalla"
    activate Web

    Web->>Plataforma: Solicita acceso a pantalla compartida
    activate Plataforma
    Plataforma-->>Web: Devuelve pantalla compartida
    deactivate Plataforma

    Web->>Web: Captura un frame de la pantalla

    Web->>API: Envía imagen capturada [HTTP POST]
    activate API

    API->>Servicio: Envía imagen para detección
    activate Servicio
    Servicio-->>API: Devuelve coordenadas de rostros
    deactivate Servicio

    API-->>Web: Retorna rostros detectados
    deactivate API

    Web-->>Usuario: Muestra rostros para seleccionar y nombrar

    Usuario->>Web: Selecciona rostros y asigna nombres

    Web->>API: Envía rostros seleccionados y nombres
    activate API

    %% Usar \n para saltos de línea en mensajes
    API->>DB: Guarda información de rostros\n(coordenadas y nombres)
    activate DB
    DB-->>API: Confirmación
    deactivate DB

    API-->>Web: Confirmación de guardado
    deactivate API
    deactivate Web
    deactivate Usuario

    %% Notas (usar \n para saltos de línea)
    note over Web,API: Transferencia segura con HTTPS
    note over Servicio: Algoritmo de detección\nfacial optimizado
